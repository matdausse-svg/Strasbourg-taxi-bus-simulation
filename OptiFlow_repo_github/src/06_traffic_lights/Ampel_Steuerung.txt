-- Hauptschleife fuer eine Kreuzung
-- Wird einmal pro Kreuzung parallel gestartet via &executeIn

param Kreuzung : integer

var Wartezeit : integer
var i : integer
var Ampel : object
var Richtung : boolean

while true

    -- Vertikal GRUEN setzen
    Kreuzung_Tabelle[1, Kreuzung] := 0
    i := 1
    while i <= Ampel_Tabelle.YDim
        if Ampel_Tabelle[3, i] = Kreuzung
            Ampel    := Ampel_Tabelle[1, i]
            Richtung := Ampel_Tabelle[2, i]
            if Richtung = false
                Ampel.Text              := "GO"
                Ampel.Hintergrundfarbe  := 65280
                Ampel.Farbe             := 0
            else
                Ampel.Text              := "STOP"
                Ampel.Hintergrundfarbe  := 255
                Ampel.Farbe             := 16777215
            end
        end
        i := i + 1
    end

    -- Mindestgruenphase vertikal
    wait 0:00:45

    -- Auf Bus horizontal warten (max 2 Minuten)
    Wartezeit := 0
    while Wartezeit < 15 and Kreuzung_Tabelle[3, Kreuzung] = 0
        wait 1s
        Wartezeit := Wartezeit + 1
    end

    -- Horizontal GRUEN setzen
    Kreuzung_Tabelle[1, Kreuzung] := 1
    i := 1
    while i <= Ampel_Tabelle.YDim
        if Ampel_Tabelle[3, i] = Kreuzung
            Ampel    := Ampel_Tabelle[1, i]
            Richtung := Ampel_Tabelle[2, i]
            if Richtung = true
                Ampel.Text              := "GO"
                Ampel.Hintergrundfarbe  := 65280
                Ampel.Farbe             := 0
            else
                Ampel.Text              := "STOP"
                Ampel.Hintergrundfarbe  := 255
                Ampel.Farbe             := 16777215
            end
        end
        i := i + 1
    end

    -- Mindestgruenphase horizontal
    wait 0:00:45

    -- Auf Bus vertikal warten (max 2 Minuten)
    Wartezeit := 0
    while Wartezeit < 15 and Kreuzung_Tabelle[2, Kreuzung] = 0
        wait 1s
        Wartezeit := Wartezeit + 1
    end

end
