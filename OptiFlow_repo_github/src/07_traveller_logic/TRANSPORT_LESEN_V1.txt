param pGruppeNummer : integer, pMaxBus : integer

var zeile, i : integer
var gefunden : boolean
var transportZufall : integer

gefunden := false
i := 1
while i <= Tabelle_Transport_Gruppe.YDim
    if Tabelle_Transport_Gruppe[1, i] = pGruppeNummer
        TransportAktuell_Global := Tabelle_Transport_Gruppe[2, i]
        gefunden := true
        i := Tabelle_Transport_Gruppe.YDim + 1
    else
        i := i + 1
    end
end

if not gefunden
    transportZufall := round(z_gleich(pGruppeNummer, 1, 10))
    if transportZufall <= pMaxBus
        TransportAktuell_Global := 1  -- Bus
    else
        TransportAktuell_Global := 0  -- Taxi
    end
    zeile := Tabelle_Transport_Gruppe.YDim + 1
    Tabelle_Transport_Gruppe[1, zeile] := pGruppeNummer
    Tabelle_Transport_Gruppe[2, zeile] := TransportAktuell_Global
end
