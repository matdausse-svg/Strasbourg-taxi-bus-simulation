-- Sensor auf Weg_Busbahnhof_Puffer_Taxi
-- Taxi oder TaxiVan wartet bis Puffer belegt ist

if @.Zielort = Weg_Busbahnhof_Puffer_Taxi
    if not @.belegt
        @.stopped := true
        while not Puffer_Taxi_Busbahnhof.belegt and not TaxiRuf_Busbahnhof
            wait 1s
        end
        @.stopped := false
        wait 1s
        if TaxiRuf_Busbahnhof
            TaxiRuf_Busbahnhof := false
            @.Zielort := TaxiRuf_Busbahnhof_Ziel
            return
        end
        BELADEN_TAXI_V2
    end
end
