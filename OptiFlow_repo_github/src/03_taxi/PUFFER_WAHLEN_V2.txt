-- ============================================================
-- Puffer_Taxi_Bahnhof zaehlen

    i := 1
    while i <= n
        beAktuell := Puffer_Taxi_Bahnhof.BE(i)
        if @.name = "Taxi"
            if beAktuell.name = "Touristen_2_4" or beAktuell.name = "Einzeltourist" or beAktuell.name = "Europaabgeordneter" or beAktuell.name = "Schueler_Student"
                nBahnhof := nBahnhof + 1
            end
        elseif @.name = "TaxiVan"
            if beAktuell.name = "Touristen_5_6" or beAktuell.name = "Touristen_2_4" or beAktuell.name = "Einzeltourist" or beAktuell.name = "Europaabgeordneter" or beAktuell.name = "Schueler_Student"

        end
        i := i + 1

end

-- ============================================================
-- Puffer_Taxi_Flughafen zaehlen
-- ============================================================

                nStadt := nStadt + 1

                nStadt := nStadt + 1

-- ============================================================

    n := Puffer_Taxi_Hotel1.numMU
    i := 1
    while i <= n
        beAktuell := Puffer_Taxi_Hotel1.BE(i)
        if @.name = "Taxi"
            if beAktuell.name = "Touristen_2_4" or beAktuell.name = "Einzeltourist" or beAktuell.name = "Europaabgeordneter" or beAktuell.name = "Schueler_Student"
                nHotel1 := nHotel1 + 1

                nHotel1 := nHotel1 + 1

        i := i + 1

if wegAktuell /= Weg_Hotel2_Puffer_Taxi
    n := Puffer_Taxi_Hotel2.numMU
    i := 1
    while i <= n
        beAktuell := Puffer_Taxi_Hotel2.BE(i)
        if @.name = "Taxi"

    maxN := nStadt

    maxN := nHotel1
end

if nHotel2 > maxN
    maxN := nHotel2
end

if nForum > maxN
    maxN := nForum
end

if nParlament > maxN
    maxN := nParlament
end

-- Wenn gar keine Kunden warten: zwischen Hubs pendeln
-- ============================================================

    WechselZaehler := WechselZaehler + 1
    if wegAktuell = Weg_Bahnhof_Puffer_Taxi
        if (WechselZaehler mod 2) = 0
            @.Zielort := Weg_Busbahnhof_Puffer_Taxi
        else
            @.Zielort := Weg_Flughafen_Puffer_Taxi
        end
    elseif wegAktuell = Weg_Busbahnhof_Puffer_Taxi
        if (WechselZaehler mod 2) = 0
            @.Zielort := Weg_Bahnhof_Puffer_Taxi
        else
            @.Zielort := Weg_Flughafen_Puffer_Taxi
        end
    elseif wegAktuell = Weg_Flughafen_Puffer_Taxi
        if (WechselZaehler mod 2) = 0
            @.Zielort := Weg_Bahnhof_Puffer_Taxi
        else
            @.Zielort := Weg_Busbahnhof_Puffer_Taxi
        end
    else
        if (WechselZaehler mod 3) = 0
            @.Zielort := Weg_Bahnhof_Puffer_Taxi
        elseif (WechselZaehler mod 3) = 1
            @.Zielort := Weg_Busbahnhof_Puffer_Taxi
        else
            @.Zielort := Weg_Flughafen_Puffer_Taxi
        end
    end

else

    WechselZaehler := WechselZaehler + 1

    if (WechselZaehler mod 8) = 0
        if nBahnhof = maxN and nBahnhof > 0
            @.Zielort := Weg_Bahnhof_Puffer_Taxi
        elseif nBusbahnhof = maxN and nBusbahnhof > 0
            @.Zielort := Weg_Busbahnhof_Puffer_Taxi
        elseif nFlughafen = maxN and nFlughafen > 0
            @.Zielort := Weg_Flughafen_Puffer_Taxi
        elseif nStadt = maxN and nStadt > 0
            @.Zielort := Weg_Stadt_Puffer_Taxi
        elseif nHotel1 = maxN and nHotel1 > 0
            @.Zielort := Weg_Hotel1_Puffer_Taxi
        elseif nHotel2 = maxN and nHotel2 > 0
            @.Zielort := Weg_Hotel2_Puffer_Taxi
        elseif nForum = maxN and nForum > 0
            @.Zielort := Weg_Forum_Puffer_Taxi
        elseif nParlament = maxN and nParlament > 0
            @.Zielort := Weg_Parlament_Puffer_Taxi
        end

        if nBusbahnhof = maxN and nBusbahnhof > 0
            @.Zielort := Weg_Busbahnhof_Puffer_Taxi

            @.Zielort := Weg_Flughafen_Puffer_Taxi
        elseif nStadt = maxN and nStadt > 0
            @.Zielort := Weg_Stadt_Puffer_Taxi
        elseif nHotel1 = maxN and nHotel1 > 0
            @.Zielort := Weg_Hotel1_Puffer_Taxi
        elseif nHotel2 = maxN and nHotel2 > 0
            @.Zielort := Weg_Hotel2_Puffer_Taxi
        elseif nForum = maxN and nForum > 0
            @.Zielort := Weg_Forum_Puffer_Taxi
        elseif nParlament = maxN and nParlament > 0
            @.Zielort := Weg_Parlament_Puffer_Taxi
        elseif nBahnhof = maxN and nBahnhof > 0
            @.Zielort := Weg_Bahnhof_Puffer_Taxi
        end

    elseif (WechselZaehler mod 8) = 2
        if nFlughafen = maxN and nFlughafen > 0
            @.Zielort := Weg_Flughafen_Puffer_Taxi
        elseif nStadt = maxN and nStadt > 0
            @.Zielort := Weg_Stadt_Puffer_Taxi
        elseif nHotel1 = maxN and nHotel1 > 0
            @.Zielort := Weg_Hotel1_Puffer_Taxi
        elseif nHotel2 = maxN and nHotel2 > 0
            @.Zielort := Weg_Hotel2_Puffer_Taxi
        elseif nForum = maxN and nForum > 0
            @.Zielort := Weg_Forum_Puffer_Taxi
        elseif nParlament = maxN and nParlament > 0
            @.Zielort := Weg_Parlament_Puffer_Taxi
        elseif nBahnhof = maxN and nBahnhof > 0
            @.Zielort := Weg_Bahnhof_Puffer_Taxi
        elseif nBusbahnhof = maxN and nBusbahnhof > 0
            @.Zielort := Weg_Busbahnhof_Puffer_Taxi
        end

    elseif (WechselZaehler mod 8) = 3
        if nStadt = maxN and nStadt > 0
            @.Zielort := Weg_Stadt_Puffer_Taxi
        elseif nHotel1 = maxN and nHotel1 > 0
            @.Zielort := Weg_Hotel1_Puffer_Taxi
        elseif nHotel2 = maxN and nHotel2 > 0
            @.Zielort := Weg_Hotel2_Puffer_Taxi
        elseif nForum = maxN and nForum > 0
            @.Zielort := Weg_Forum_Puffer_Taxi
        elseif nParlament = maxN and nParlament > 0
            @.Zielort := Weg_Parlament_Puffer_Taxi
        elseif nBahnhof = maxN and nBahnhof > 0
            @.Zielort := Weg_Bahnhof_Puffer_Taxi
        elseif nBusbahnhof = maxN and nBusbahnhof > 0
            @.Zielort := Weg_Busbahnhof_Puffer_Taxi
        elseif nFlughafen = maxN and nFlughafen > 0
            @.Zielort := Weg_Flughafen_Puffer_Taxi
        end

    elseif (WechselZaehler mod 8) = 4
        if nHotel1 = maxN and nHotel1 > 0
            @.Zielort := Weg_Hotel1_Puffer_Taxi
        elseif nHotel2 = maxN and nHotel2 > 0
            @.Zielort := Weg_Hotel2_Puffer_Taxi
        elseif nForum = maxN and nForum > 0
            @.Zielort := Weg_Forum_Puffer_Taxi
        elseif nParlament = maxN and nParlament > 0
            @.Zielort := Weg_Parlament_Puffer_Taxi

        end

        if nParlament = maxN and nParlament > 0
            @.Zielort := Weg_Parlament_Puffer_Taxi
        elseif nBahnhof = maxN and nBahnhof > 0
            @.Zielort := Weg_Bahnhof_Puffer_Taxi
        elseif nBusbahnhof = maxN and nBusbahnhof > 0
            @.Zielort := Weg_Busbahnhof_Puffer_Taxi

            @.Zielort := Weg_Flughafen_Puffer_Taxi

            @.Zielort := Weg_Stadt_Puffer_Taxi
