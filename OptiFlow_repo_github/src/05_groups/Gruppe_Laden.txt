-- Laedt alle BEs einer Gruppe aus dem Puffer in das Fahrzeug (@)

param pPuffer : object, pGruppenNummer : integer

var n, i : integer
var beAktuell : object

n := pPuffer.numMU
i := 1

while i <= n
    beAktuell := pPuffer.BE(i)
    if beAktuell.Gruppe_Nummer = pGruppenNummer
        beAktuell.umlagern(@)
        n := pPuffer.numMU
        i := 1
    else
        i := i + 1
    end
end
